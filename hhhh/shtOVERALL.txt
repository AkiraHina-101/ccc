Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo done
    If Intersect(Target, Union(Me.Range("D2:D3"), Me.Range("AQ13"), _
                              Me.Range("AQ32"), Me.Range("N3:N6"), _
                              Me.Range("S3:S6"))) Is Nothing Then Exit Sub

    Application.EnableEvents = False
    If Not Intersect(Target, Union(Me.Range("D2"), Me.Range("AQ13"))) Is Nothing Then
        Me.Range("AR13:CF28").Calculate
    End If
    If Not Intersect(Target, Union(Me.Range("D3"), Me.Range("AQ32"))) Is Nothing Then
        Me.Range("AR32:CF32").Calculate
        Me.Range("AS220:CF225").Calculate
    End If
    If Not Intersect(Target, Union(Me.Range("D2:D3"), Me.Range("AQ13"), _
                                   Me.Range("AQ32"))) Is Nothing Then _
        modOverall.RefreshOverallSeriesRanges
    If Not Intersect(Target, Me.Range("N3:N6")) Is Nothing Then
        ApplyOverallYAxis Array("chOverallMulti_1", "chOverallMulti_2", _
                                "chOverallMulti_3", "chOverallMulti_4"), _
                          Me.Range("N3:N6")
    End If
    If Not Intersect(Target, Me.Range("S3:S6")) Is Nothing Then
        ApplyOverallYAxis Array("chOverallSingle_1", "chOverallSingle_2", _
                                "chOverallSingle_3", "chOverallSingle_4"), _
                          Me.Range("S3:S6")
    End If
done:
    Application.EnableEvents = True
End Sub

' Applies Min, Max, Major and Minor values to one OVERALL chart group.
Private Sub ApplyOverallYAxis(ByVal chartNames As Variant, _
                              ByVal controlCells As Range)
    Dim chartName As Variant, targetAxis As Axis

    For Each chartName In chartNames
        Set targetAxis = Me.ChartObjects(CStr(chartName)).Chart.Axes(xlValue)
        On Error Resume Next
        targetAxis.MinimumScaleIsAuto = True
        targetAxis.MaximumScaleIsAuto = True
        targetAxis.MajorUnitIsAuto = True
        targetAxis.MinorUnitIsAuto = True
        If IsNumeric(controlCells.Cells(1, 1).Value2) And _
           Len(controlCells.Cells(1, 1).Value2 & "") > 0 Then _
            targetAxis.MinimumScale = CDbl(controlCells.Cells(1, 1).Value2)
        If IsNumeric(controlCells.Cells(2, 1).Value2) And _
           Len(controlCells.Cells(2, 1).Value2 & "") > 0 Then _
            targetAxis.MaximumScale = CDbl(controlCells.Cells(2, 1).Value2)
        If IsNumeric(controlCells.Cells(3, 1).Value2) And _
           controlCells.Cells(3, 1).Value2 > 0 Then _
            targetAxis.MajorUnit = CDbl(controlCells.Cells(3, 1).Value2)
        If IsNumeric(controlCells.Cells(4, 1).Value2) And _
           controlCells.Cells(4, 1).Value2 > 0 Then _
            targetAxis.MinorUnit = CDbl(controlCells.Cells(4, 1).Value2)
        On Error GoTo 0
    Next chartName
End Sub
