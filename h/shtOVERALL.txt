Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo done
    If Intersect(Target, Me.Range("D2:D3")) Is Nothing Then Exit Sub

    Application.EnableEvents = False
    If Not Intersect(Target, Me.Range("D2")) Is Nothing Then
        Me.Range("AR13:CF24").Calculate
    End If
    If Not Intersect(Target, Me.Range("D3")) Is Nothing Then
        Me.Range("AR32:CF32").Calculate
        Me.Range("AS220:CF225").Calculate
    End If
done:
    Application.EnableEvents = True
End Sub
