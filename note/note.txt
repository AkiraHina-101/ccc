Option Explicit

' Quantity changes update the Y title; edits in the small scale block update
' all chart axes immediately.
Private Sub Worksheet_Change(ByVal target As Range)
    Dim scaleCells As Range
    On Error GoTo done
    If Not Intersect(target, Me.Range("selQty")) Is Nothing Then
        modRebuild.ApplyQuantityAxes
    End If
    Set scaleCells = Union(Me.Range("axXmin").Resize(4, 1), _
                           Me.Range("axYmin").Resize(4, 1))
    If Not Intersect(target, scaleCells) Is Nothing Then
        Application.EnableEvents = False
        modRebuild.ApplyConfiguredAxes
    End If
done:
    Application.EnableEvents = True
End Sub
