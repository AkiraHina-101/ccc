"""Build a layered TET4 mesh conforming to Cylinder_1's saved surface mesh.

Input comes from ``export_cylinder_surface.py``. The script never edits the
original Jupiter model or its DLLs. It produces a Nastran BDF and validates
that the tetra boundary equals the existing TRI3 boundary exactly by node ID.
"""

from collections import Counter
import json
from pathlib import Path


ROOT = Path(__file__).resolve().parent.parent
SOURCE = ROOT / "cylinder_surface_snapshot.json"
OUTPUT = ROOT / "Cylinder_1_layered_tetra.bdf"
SOLID_ONLY_OUTPUT = ROOT / "Cylinder_1_layered_tetra_solid_only.bdf"


def key_xz(pos):
    return (round(pos[0], 10), round(pos[2], 10))


def signed_six_volume(a, b, c, d):
    ax, ay, az = (b[i] - a[i] for i in range(3))
    bx, by, bz = (c[i] - a[i] for i in range(3))
    cx, cy, cz = (d[i] - a[i] for i in range(3))
    return ax * (by * cz - bz * cy) - ay * (bx * cz - bz * cx) + az * (bx * cy - by * cx)


def faces_of_tet(tet):
    a, b, c, d = tet
    return (
        tuple(sorted((a, b, c))),
        tuple(sorted((a, b, d))),
        tuple(sorted((a, c, d))),
        tuple(sorted((b, c, d))),
    )


def build(source):
    nodes = {int(k): tuple(v) for k, v in source["nodes"].items()}
    surface = [(item["id"], tuple(item["nodes"])) for item in source["triangles"]]
    original_nodes = set(nodes)
    original_surface = Counter(tuple(sorted(tri)) for _, tri in surface)
    levels = sorted({round(pos[1], 10) for pos in nodes.values()})
    if len(levels) < 2:
        raise ValueError("Surface has fewer than two axial levels")
    lower, upper = levels[0], levels[-1]

    caps = {}
    for y in (lower, upper):
        caps[y] = [tri for _, tri in surface if all(abs(nodes[n][1] - y) < 1e-9 for n in tri)]
    if not caps[lower] or len(caps[lower]) != len(caps[upper]):
        raise ValueError("Top and bottom cap meshes do not match in size")
    projected = lambda tri: tuple(sorted(key_xz(nodes[n]) for n in tri))
    if Counter(map(projected, caps[lower])) != Counter(map(projected, caps[upper])):
        raise ValueError("Top and bottom triangulations are not congruent")

    base_tris = [tuple(key_xz(nodes[n]) for n in tri) for tri in caps[lower]]
    edge_count = Counter(
        tuple(sorted((tri[i], tri[(i + 1) % 3])))
        for tri in base_tris for i in range(3)
    )
    perimeter = {edge for edge, count in edge_count.items() if count == 1}
    if any(count not in (1, 2) for count in edge_count.values()):
        raise ValueError("Cap triangulation is nonmanifold")

    existing = {}
    for nid, pos in nodes.items():
        lookup = (round(pos[1], 10), key_xz(pos))
        if lookup in existing and existing[lookup] != nid:
            raise ValueError("Coincident boundary nodes: {}".format(lookup))
        existing[lookup] = nid

    next_node = max(nodes) + 1
    layer_node = {}
    base_positions = sorted({point for tri in base_tris for point in tri})
    for y in levels:
        for xz in base_positions:
            lookup = (y, xz)
            if lookup in existing:
                layer_node[lookup] = existing[lookup]
            elif y not in (lower, upper) and xz not in {p for edge in perimeter for p in edge}:
                layer_node[lookup] = next_node
                nodes[next_node] = (xz[0], y, xz[1])
                next_node += 1
            else:
                raise ValueError("Missing boundary node at {}".format(lookup))

    side_triangles = {}
    for _, tri in surface:
        ys = {round(nodes[n][1], 10) for n in tri}
        if len(ys) != 2:
            continue
        xzs = frozenset(key_xz(nodes[n]) for n in tri)
        if len(xzs) != 2:
            raise ValueError("Unexpected side triangle {}".format(tri))
        side_triangles.setdefault((tuple(sorted(ys)), xzs), []).append(tri)

    tets = []
    min_six_volume = float("inf")
    for level_index in range(len(levels) - 1):
        lo, hi = levels[level_index : level_index + 2]
        for base in base_tris:
            low = [layer_node[(lo, point)] for point in base]
            high = [layer_node[(hi, point)] for point in base]
            boundary = [tuple(low), tuple(high)]
            for edge_index in range(3):
                p = base[edge_index]
                q = base[(edge_index + 1) % 3]
                edge = tuple(sorted((p, q)))
                if edge in perimeter:
                    sides = side_triangles.get(((lo, hi), frozenset((p, q))), [])
                    if len(sides) != 2:
                        raise ValueError("Side quad does not contain exactly two triangles")
                    boundary.extend(sides)
                else:
                    a, b = edge
                    al, bl = layer_node[(lo, a)], layer_node[(lo, b)]
                    ah, bh = layer_node[(hi, a)], layer_node[(hi, b)]
                    boundary.extend(((al, bl, bh), (al, bh, ah)))

            if len(boundary) != 8:
                raise AssertionError("A triangular prism must have eight triangular facets")
            coords = [nodes[n] for n in low + high]
            nodes[next_node] = tuple(sum(pos[i] for pos in coords) / 6 for i in range(3))
            center = next_node
            next_node += 1
            for face in boundary:
                a, b, c = face
                tet = (center, a, b, c)
                volume = signed_six_volume(*(nodes[n] for n in tet))
                if abs(volume) < 1e-18:
                    raise ValueError("Degenerate tetrahedron")
                if volume < 0:
                    tet = (center, b, a, c)
                    volume = -volume
                min_six_volume = min(min_six_volume, volume)
                tets.append(tet)

    face_counts = Counter(face for tet in tets for face in faces_of_tet(tet))
    if any(count > 2 for count in face_counts.values()):
        raise ValueError("Generated tetrahedra contain nonmanifold faces")
    generated_surface = Counter({face: 1 for face, count in face_counts.items() if count == 1})
    if generated_surface != original_surface:
        missing = original_surface - generated_surface
        extra = generated_surface - original_surface
        raise ValueError("Boundary mismatch: {} missing, {} extra".format(sum(missing.values()), sum(extra.values())))
    if len(original_nodes - set(nodes)):
        raise AssertionError("Original nodes disappeared")
    return nodes, surface, tets, levels, min_six_volume


def write_bdf(path, nodes, surface, tets, include_surface=True):
    # Small-field free format is supported by Jupiter's Nastran BDF importer.
    with path.open("w", newline="\n") as stream:
        stream.write("$ Cylinder_1 layered TET4 mesh; original surface node IDs retained\n")
        stream.write("BEGIN BULK\n")
        stream.write("MAT1,1,2.1E11,,0.3\n")
        if include_surface:
            stream.write("PSHELL,1,1,0.001\n")
        stream.write("PSOLID,2,1\n")
        # Jupiter's Nastran importer interprets GRID coordinates as mm while
        # the live model reports SI metres. Multiply by 1000 for round-trip.
        for nid, (x, y, z) in sorted(nodes.items()):
            stream.write("GRID,{},{},{:.12g},{:.12g},{:.12g}\n".format(
                nid, "", x * 1000, y * 1000, z * 1000
            ))
        if include_surface:
            for eid, tri in surface:
                stream.write("CTRIA3,{},1,{}, {}, {}\n".format(eid, *tri).replace(" ", ""))
        next_eid = max(eid for eid, _ in surface) + 1
        for index, tet in enumerate(tets):
            stream.write("CTETRA,{},2,{}, {}, {}, {}\n".format(next_eid + index, *tet).replace(" ", ""))
        stream.write("ENDDATA\n")


def main():
    with SOURCE.open() as stream:
        source = json.load(stream)
    nodes, surface, tets, levels, min_six_volume = build(source)
    write_bdf(OUTPUT, nodes, surface, tets)
    write_bdf(SOLID_ONLY_OUTPUT, nodes, surface, tets, include_surface=False)
    print("levels={} surface_TRI3={} TET4={} nodes={} min_tet_volume={:.9g} BDF={}".format(
        len(levels), len(surface), len(tets), len(nodes), min_six_volume / 6, OUTPUT
    ))


if __name__ == "__main__":
    main()
