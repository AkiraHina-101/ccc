hwi OpenStack
hwi GetSessionHandle sess

set outfile [tk_getSaveFile \
    -title "Save HyperView GIF" \
    -defaultextension ".gif" \
    -filetypes {{"GIF Animation" {.gif}}}]

if {$outfile eq ""} {
    hwi CloseStack
    return
}

sess CaptureAnimation \
    gif \
    $outfile \
    pixels 1920 1080
