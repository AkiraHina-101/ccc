package require postquery

# ============================================================
# USER SETTINGS
# ============================================================

# Enter the exact Saved View names to export
set views {"View 1" "View 2" "View 3" "View 4"}

# Fixed GIF output size
set width 1280
set height 720


# ============================================================
# GET CURRENT SIMULATION TITLE
# ============================================================

if {[catch {set simulation_title [::hwp::getCurrentSimulation -type name]}]} {
    set simulation_title "HyperView_Capture"
}

if {[string trim $simulation_title] eq ""} {
    set simulation_title "HyperView_Capture"
}

# Replace characters that are not valid in Windows file names
regsub -all {[/\\:*?"<>|]} $simulation_title "_" safe_simulation_name
set safe_simulation_name [string trim $safe_simulation_name " ."]


# ============================================================
# SELECT OUTPUT DIRECTORY
# ============================================================

# Remember the last output directory during the current session
if {![info exists ::HV_LastGifDir] || ![file isdirectory $::HV_LastGifDir]} {
    set ::HV_LastGifDir [pwd]
}

set output_dir [tk_chooseDirectory \
    -title "Select GIF Output Folder" \
    -initialdir $::HV_LastGifDir]

# Exit if the user cancels
if {$output_dir eq ""} {
    return
}

# Remember the selected directory
set ::HV_LastGifDir $output_dir


# ============================================================
# GET HYPERVIEW HANDLES
# ============================================================

hwi OpenStack

hwi GetSessionHandle session_handle
session_handle GetProjectHandle project_handle

set page_id [project_handle GetActivePage]
project_handle GetPageHandle page_handle $page_id

set window_id [page_handle GetActiveWindow]
page_handle GetWindowHandle window_handle $window_id

window_handle GetClientHandle post_handle
window_handle GetViewControlHandle view_handle


# ============================================================
# EXPORT EACH SAVED VIEW
# ============================================================

foreach view_name $views {

    # Activate the requested Saved View
    if {[catch {
        view_handle SetActiveView true $view_name
    } view_error]} {
        puts "WARNING: Cannot activate view '$view_name'. Skipped."
        continue
    }

    # Redraw the HyperView window before capture
    post_handle Draw
    update idletasks

    # Create a Windows-safe view name
    set safe_view_name $view_name
    regsub -all {[/\\:*?"<>|]} $safe_view_name "_" safe_view_name
    set safe_view_name [string trim $safe_view_name " ."]

    # Build the automatic GIF file name
    set outfile [file join $output_dir \
        "${safe_simulation_name}_${safe_view_name}.gif"]

    puts "Exporting: $outfile"

    # Capture the full current HyperView content
    session_handle CaptureAnimation \
        gif \
        $outfile \
        pixels $width $height
}

hwi CloseStack
