hwi OpenStack
hwi GetSessionHandle sess

set outfile [tk_getSaveFile \
    -title "Save HyperView GIF" \
    -defaultextension ".gif" \
    -filetypes {{"GIF Animation" {.gif}}}]

if {$outfile eq ""} {
    hwi CloseStack
    return
}

# Fixed output frame size
set width 1280
set height 720

# Keep full current HyperView page/graphics content
sess CaptureAnimation \
    gif \
    $outfile \
    pixels $width $height

hwi CloseStack
