# ============================================================
# SELECT OUTPUT DIRECTORY
# ============================================================

# Remember the last output directory during the current session
if {![info exists ::HV_LastGifDir] || ![file isdirectory $::HV_LastGifDir]} {
    set ::HV_LastGifDir [pwd]
}

# Use the standard Save dialog to select the output location
set selected_path [tk_getSaveFile \
    -title "Select GIF Output Location" \
    -initialdir $::HV_LastGifDir \
    -initialfile "${safe_simulation_name}.gif" \
    -defaultextension ".gif" \
    -filetypes {{"GIF Animation" {.gif}}}]

# Exit if the user cancels
if {$selected_path eq ""} {
    return
}

# Use only the selected folder
set output_dir [file dirname $selected_path]

# Remember the selected directory for the next run
set ::HV_LastGifDir $output_dir
