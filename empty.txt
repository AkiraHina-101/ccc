Option Explicit

Private gFormulas() As Variant
Private gHasFormula() As Boolean
Private gRows As Long
Private gCols As Long
Private gFormulaCopied As Boolean


Sub Copy_Formulas()

    Dim src As Range
    Dim i As Long
    Dim j As Long

    On Error Resume Next
    Set src = Application.InputBox( _
        Prompt:="Select the range to copy formulas from:", _
        Title:="Copy Formulas", _
        Type:=8)
    On Error GoTo 0

    If src Is Nothing Then Exit Sub

    gRows = src.Rows.Count
    gCols = src.Columns.Count

    ReDim gFormulas(1 To gRows, 1 To gCols)
    ReDim gHasFormula(1 To gRows, 1 To gCols)

    For i = 1 To gRows
        For j = 1 To gCols

            If src.Cells(i, j).HasFormula Then

                ' Store formula in R1C1 format
                ' so relative references move correctly when pasted
                gFormulas(i, j) = src.Cells(i, j).FormulaR1C1
                gHasFormula(i, j) = True

            Else

                gHasFormula(i, j) = False

            End If

        Next j
    Next i

    gFormulaCopied = True

    MsgBox _
        "Formula range copied." & vbCrLf & vbCrLf & _
        "Size: " & gRows & " rows x " & gCols & " columns", _
        vbInformation, _
        "Copy Formulas"

End Sub


Sub Paste_Formulas()

    Dim dst As Range
    Dim startCell As Range
    Dim i As Long
    Dim j As Long
    Dim formulaCount As Long

    If Not gFormulaCopied Then
        MsgBox "No formula range has been copied yet.", _
               vbExclamation, "Paste Formulas"
        Exit Sub
    End If

    On Error Resume Next
    Set dst = Application.InputBox( _
        Prompt:="Select the START cell for pasting formulas:", _
        Title:="Paste Formulas", _
        Type:=8)
    On Error GoTo 0

    If dst Is Nothing Then Exit Sub

    Set startCell = dst.Cells(1, 1)

    Application.ScreenUpdating = False

    On Error GoTo PasteError

    For i = 1 To gRows
        For j = 1 To gCols

            If gHasFormula(i, j) Then

                startCell.Offset(i - 1, j - 1).FormulaR1C1 = _
                    gFormulas(i, j)

                formulaCount = formulaCount + 1

            End If

        Next j
    Next i

    Application.ScreenUpdating = True

    MsgBox _
        formulaCount & " formulas pasted." & vbCrLf & vbCrLf & _
        "Start cell: " & startCell.Address(False, False), _
        vbInformation, _
        "Paste Formulas"

    Exit Sub


PasteError:

    Application.ScreenUpdating = True

    MsgBox _
        "Paste failed:" & vbCrLf & Err.Description, _
        vbCritical, _
        "Paste Formulas"

End Sub
