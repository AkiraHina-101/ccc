from pyjdg import *

def main():
    dlg=JDGCreator(title="ダイアログ",resizable=True,validation=True)
    dlg.add_table(name="Table1",rows=5,
        columns=["見出し1","見出し2"],
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="OK",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_table_right_menu(name="Table1",
        menus=["セルの色を設定",
            "テキストの色を設定",
            "列幅を設定",
            "カスタムメニュー"])
    dlg.generate_window()

if __name__=='__main__':
    main()
# Encoding for Japanese
# coding: cp932

from pyjdg import *

def on_menu(dlg,name,menu):
    print("テーブル名: "+ name)
    print("クリックされたメニュー: "+ menu)
    cellvector = dlg.get_table_sel_cell(name="Table1")
    if menu == "列幅を設定":
        dlg.set_table_column_width(name="Table1",
        col=cellvector[0].col_number)
    elif menu == "テキストの色を設定":
        dlg.set_table_cell_text_color(name="Table1",
        cell=cellvector[0])
    elif menu == "セルの色を設定":
        dlg.set_table_cell_fill_color(name="Table1")
    elif menu == "カスタムメニュー":
        print("ここに自分の関数を置くことができます")

def main():
    dlg=JDGCreator(title="ダイアログ",resizable=True,validation=True)
    dlg.add_table(name="Table1",rows=5,
        columns=["見出し1","見出し2"],
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="OK",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_table_right_menu(name="Table1",
        menus=["セルの色を設定","テキストの色を設定",
        "列幅を設定","カスタムメニュー"])
    dlg.generate_window()
    dlg.on_table_right_menu(name="Table1",callfunc=on_menu)

if __name__=='__main__':
    main()
