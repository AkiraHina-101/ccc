テーブルの列データ型

🎯 はじめに​

このチュートリアルでは、テーブルの入力データ型を制御する方法を学びます。

  1. テーブルを作成します。
  2. テーブルの列データ型を設定します。

無効なデータが入力されると、テーブルセルには内容が表示されません。

📖 チュートリアル​

# Encoding for Japanese
# coding: cp932

from pyjdg import *

def main():
    dlg=JDGCreator(title="ダイアログ",resizable=True,validation=True)
    dlg.add_table(name="Table1",rows=5,
        columns=["文字列","整数","倍精度浮動小数点数"],
        layout="Window",width=350,height=260)
    dlg.set_table_column_data_type(name="Table1",
        col=0,data_type="String")
    dlg.set_table_column_data_type(name="Table1",
        col=1,data_type="Integer")
    dlg.set_table_column_data_type(name="Table1",
        col=2,data_type="Double",precision=5)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="Ok",text="OK",layout="footer")
    dlg.add_button(name="ButtonCancel",text="キャンセル",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()

if __name__=='__main__':
    main()
