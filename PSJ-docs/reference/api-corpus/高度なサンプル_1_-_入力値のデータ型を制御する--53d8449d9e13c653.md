# 高度なサンプル 1 - 入力値のデータ型を制御する

高度なサンプル 1 - 入力値のデータ型を制御する

🎯 はじめに​

このチュートリアルでは、テーブルが入力値のデータ型を制御し、イベントをトリガーする方法を学びます：

空のセルをチェックする: 空のセルには赤色を塗り、エラーメッセージを表示します。

セルのデータ型をチェックする: 「String」列には文字を、「Integer」列には数値を入力する必要があります。

「String」列のセルに数値が入力された場合、JPTは赤色で塗り、エラーメッセージを表示します。

「Integer」列のセルに文字が入力された場合、JPTはオレンジ色で塗り、エラーメッセージを表示します。

📖 チュートリアル​

# Python APIウィンドウのエンコーディング
# coding: cp932

# 'sys' モジュールのインポート
import sys

# Python APIウィンドウのログメッセージをクリア
JPT.ClearLog()

from pyjdg import *

def on_click_check(dlg):
    '''
    すべてのセルをループし、各セルに対して2つのチェックを実行します：
    1. セルが空かどうかを確認し、セルが空の場合は赤で塗り、入力を促します。
        引数:
            なし
        戻り値:
            空の赤いセル
    2. セルのデータ検証をチェックし、データ型が正しくない場合は塗りつぶしセルの色を変更します。
        引数:
            なし
        戻り値:
            値が文字列でない場合は赤いセル
            値が数値でない場合はオレンジのセル
    '''
    JPT.ClearLog()
    total_rows = dlg.get_total_row(name="Table1")
    total_cols = dlg.get_total_column(name="Table1")
    i = 0
    j = 0
    k = 0
    # 必須入力のチェック
    for row in range(total_rows):
        for col in range(total_cols):
            value_cell = dlg.get_cell_value(name="Table1", row=row, col=col)
            if value_cell:
                dlg.set_table_cell_fill_color(name="Table1", cell=TableCellID(row=row, col=col), color=16777215)
            elif not value_cell:
                dlg.set_table_cell_fill_color(name="Table1", cell=TableCellID(row=row, col=col), color=7105764)
                k += 1
                continue
    if k != 0:
        JPT.MsgOut("赤いセルは入力が必要です")
        return
    else:
        # データ検証のチェック
        for row in range(total_rows):
            for col in range(total_cols):
                value_cell = dlg.get_cell_value(name="Table1", row=row, col=col)
                dlg.set_table_cell_fill_color(name="Table1", cell=TableCellID(row=row, col=col), color=16777215)
                if col == 0:
                    if value_cell.isnumeric() == True:
                        dlg.set_table_cell_fill_color(name="Table1", cell=TableCellID(row=row, col=col), color=7105764)
                        i += 1

                elif col == 1:
                    if value_cell.isnumeric() == False:
                        dlg.set_table_cell_fill_color(name="Table1", cell=TableCellID(row=row, col=col), color=42495)
                        j += 1

    if i != 0 and j != 0:
        JPT.MsgOut("赤いセルは文字列型が必要です")
        JPT.MsgOut("オレンジのセルは整数型が必要です")
    elif i != 0 and j == 0:
        JPT.MsgOut("赤いセルは文字列型が必要です")
    elif i == 0 and j != 0:
        JPT.MsgOut("オレンジのセルは整数型が必要です")

def main():
    dlg=JDGCreator(title="ダイアログ", resizable=True, validation=True)
    dlg.add_table(name="Table1", rows=5,
        columns=["String", "Integer"],
        layout="Window", menus=["clear"], width=260, height=260)
    dlg.set_table_column_data_type(name="Table1", col=0, data_type="String")
    dlg.add_hlayout(name="footer", layout="Window")
    dlg.add_space(orientation="horizontal", layout="footer")
    dlg.add_button(name="ButtonApply", text="適用", layout="footer")
    dlg.add_button(name="ButtonOk", text="OK", layout="footer")
    dlg.add_button(name="ButtonCancel", text="キャンセル", layout="footer")
    dlg.add_space(orientation="horizontal", layout="footer")
    dlg.generate_window()
    dlg.on_button_clicked(name="ButtonApply", callfunc=on_click_check)

if __name__=='__main__':
    main()
