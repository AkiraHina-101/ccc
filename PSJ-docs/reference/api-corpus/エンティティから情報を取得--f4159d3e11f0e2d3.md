# エンティティから情報を取得

エンティティから情報を取得

🎯 はじめに​

本チュートリアルでは、エンティティからデータベース情報を取得する方法を紹介します:

インストールされているJupiterのパスを取得し、サンプルモデルをJupiterに読み込む。

モデルデータベースからエンティティ(ローカル設定-フェース/境界条件-圧力/グループ)のリストを取得し、各エンティティの名前、ID、エンティティタイプ、ターゲットIDを表示する。

モデルデータベースからターゲットエンティティのIDごとにエンティティ(要素/フェース)のリストを取得し、エンティティ毎に色を変更する。

📖 チュートリアル​

# Python APIウィンドウ内のログメッセージを消去します。
JPT.ClearLog()

# インストールされているJupiterとサンプルデータのパスを取得します。その後サンプルモデルをJupiterに読み込みます。
jupiterPath = JPT.GetAppPathInfo(JPT.PathType.PROGRAM_PATH)
samplePath = jupiterPath + '/SampleData/PSJ/PSJ-Utility/JtdbSample/EntityModel.jtdb'
FileMenu.LoadJTDB(samplePath)
JPT.ViewFitToModel()

# モデルデータベースから'ローカル設定-フェース'のエンティティリストを取得します。
# 次にローカル設定の名前、ID、エンティティタイプおよびすべてのフェースIDを表示します。
listLocalSetting = JPT.GetAllByType(JPT.DItemType.LOCAL_SETTING)
if listLocalSetting.sizeVec() > 0:
    for ls in listLocalSetting:
        print ("ローカル設定-名前: " + ls.name)
        print ("ローカル設定-ID: " + str(ls.id))
        print ("エンティティタイプ: " + str(ls.type))
        listTargetFaces = ls.targets
        print ("ローカル設定-フェースのターゲットフェースID一覧:")
        for face in listTargetFaces:
            print(face.id)

# モデルデータベースから'圧力-一般'のエンティティリストを取得します。
# 次に圧力-一般の名前、ID、エンティティタイプおよびすべてのフェースIDを表示します。
listGeneralPressure = JPT.GetAllByType(JPT.DItemType.LBC_G_PRESSURE)
if listGeneralPressure.sizeVec() > 0:
    for pressure in listGeneralPressure:
        print ("圧力-名前: " + pressure.name)
        print ("圧力-ID: " + str(pressure.id))
        print ("エンティティタイプ: " + str(pressure.type))
        listTargetFaces = pressure.targets
        print ("圧力-一般のターゲットフェースID一覧:")
        for face in listTargetFaces:
            print(face.id)

# モデルデータベースから'グループ'のエンティティリストを取得します。
# 次にグループの名前、ID、エンティティタイプおよび要素IDを表示します。
listGroup = JPT.GetAllByType(JPT.DItemType.GROUP)
if listGroup.sizeVec() > 0:
    for group in listGroup:
        print ("グループ-名前: " + group.name)
        print ("グループ-ID: " + str(group.id))
        print ("エンティティタイプ: " + str(group.type))
        listTargetElems = group.targets
        print ("グループのターゲット要素ID一覧:")
        for elem in listTargetElems:
            print(elem.id)

# モデルデータベースからID:239の'要素'エンティティのリストを取得します。
# 次に要素ID:239の色を白に変更します。
listElemEntityObject = JPT.GetEntitiesByID(JPT.DItemType.ELEM, 239)
if listElemEntityObject.sizeVec() == 1:
    elemObject = JPT.CastDItemToDFace(listElemEntityObject[0])
    elemObject.color = JPT.ConvertRGBToJPTColor(255, 255, 255)

# モデルデータベースからID-23の'フェース'エンティティのリストを取得します。
# 次にフェースID:23の色を緑に変更します。
listFaceEntityObject = JPT.GetEntitiesByID(JPT.DItemType.FACE, 23)
if listFaceEntityObject.sizeVec() == 1:
    faceObject = JPT.CastDItemToDFace(listFaceEntityObject[0])
    faceObject.color = JPT.ConvertRGBToJPTColor(0, 255, 0)
