このチュートリアルでは、次の内容を含むシンプルなウィザード (PagesCtrl)ダイアログの作成方法を学びます：

最初のページには、ラベルとテキストボックスがあります。

2ページ目には、コンボボックスとリストボックスがあります。

📖 チュートリアル​

# Python APIウィンドウのエンコーディング
# coding: cp932

from pyjdg import *

def main():
    dlg=JDGCreator(title="PagesCtrl",resizable=True,validation=True,include_apply=False)
    dlg.add_pagesctrl(name="PagesCtrl2",width=300,height=300,current_page=0,layout="Window")
    dlg.add_pageitem(name="PagesCtrl2",page_name="PageItem3",page_header="PageItem")
    dlg.add_label(name="Label5",text="Label",text_halign="left",text_valign="top",layout="PageItem3")
    dlg.add_textbox(name="TextBox6",layout="PageItem3")
    dlg.add_pageitem(name="PagesCtrl2",page_name="PageItem4",page_header="PageItem")
    dlg.add_combobox(name="ComboBox7",layout="PageItem4")
    dlg.add_listbox(name="ListBox8",width=100,height=120,layout="PageItem4")
    dlg.set_item_size_behavior(name="ListBox8",behavior=size_behavior.fixed)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="Ok",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()

if __name__=='__main__':
    main()
