# 列のフィルタを表示/非表示にする

列のフィルタを表示/非表示にする

🎯 はじめに​

このチュートリアルでは、テーブルの列のフィルタオプションを表示または非表示にする方法を学びます：

TableColumnInfo() でフィルタを表示/非表示にする: TableColumnInfo() オブジェクトを制御してテーブルの列のフィルタを表示および非表示にします。

dlg.enable_table_column_filter() でフィルタを表示/非表示にする: API dlg.enable_table_column_filter() を使用してテーブルの列のフィルタを表示および非表示にします。

TableColumnInfo() でフィルタを表示/非表示にする

dlg.enable_table_column_filter() でフィルタを表示/非表示にする

📖 チュートリアル​

TableColumnInfo() でフィルタを表示/非表示にする

dlg.enable_table_column_filter() でフィルタを表示/非表示にする

# 日本語のエンコーディング
# coding: cp932

from pyjdg import *

def main():
    dlg=JDGCreator(title="Dialog",resizable=True,validation=True)
    column_vector=TableColumnInfoVector()
    column_vector.add(TableColumnInfo(name="Heading1",type="string",filter=True))
    column_vector.add(TableColumnInfo(name="Heading2",type="string",filter=True))
    dlg.add_table(name="Table2",rows=5,
        columns=column_vector,
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="Ok",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()

if __name__=='__main__':
    main()

# coding: cp932

from pyjdg import *

def main():
    dlg=JDGCreator(title="Dialog",resizable=True,validation=True)
    dlg.add_table(name="Table2",rows=5,
        columns=["Heading1","Heading2"],
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="Ok",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.enable_table_column_filter(name="Table2",col=0,enable=True)
    dlg.enable_table_column_filter(name="Table2",col=1,enable=True)
    dlg.generate_window()

if __name__=='__main__':
    main()
