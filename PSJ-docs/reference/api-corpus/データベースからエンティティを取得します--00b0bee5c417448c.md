# データベースからエンティティを取得します

データベースからエンティティを取得します

🎯 導入​

このチュートリアルでは、データベースからエンティティを取得する方法を学びます。

特定のエンティティのリストを取得します（Part/Face/Edge/Element/Node/Group/Connection/Coordinate/LBC）。

エンティティのリストを取得しますテーブルタイプ（ロードケース/ローカルセット）。

エンティティのタイプ（Edge/1D要素）ごとにエンティティのリストを取得します。

プロパティのIDで資料を取得します。

パートのIDでパートに関連付けられたすべての顔のリストを取得します。

エンティティのタイプとエンティティのプロパティ（名前、ID）でエンティティのリストを取得します。

得るNodex、y、z位置によるエンティティ。

📖 チュートリアル​

# Python APIウィンドウでログメッセージをクリアします
JPT.ClearLog()

# モデルを準備します
path = JPT.GetAppPathInfo(JPT.PathType.PROGRAM_PATH)
FileMenu.LoadJTDB(path + "/SampleData/PSJ/PSJ-Utility/JtdbSample/Get_Infor_Part_Sample.jtdb")
JPT.ViewFitToModel()

############ データベースから特定のエンティティのリストを取得します ##############
# すべての「部分」エンティティのリストを取得します
listParts = JPT.GetAllParts()
JPT.Debugger(listParts) # リスト Body, サイズ = 3

# すべての「フェイス」エンティティのリストを取得します
listFaces = JPT.GetAllFaces()
JPT.Debugger(listFaces) # リストFace, サイズ = 26

# すべての「エッジ」エンティティのリストを取得します
listEdges = JPT.GetAllEdges()
JPT.Debugger(listEdges) # リスト Edge, サイズ = 173

# すべての「要素」エンティティのリストを取得します
listElems = JPT.GetAllElems()
JPT.Debugger(listElems) # リスト Elem, サイズ = 12595

# すべての「節点」エンティティのリストを取得します
listNodes = JPT.GetAllNodes()
JPT.Debugger(listNodes) # リスト Node, サイズ = 15401

# すべての「グループ」エンティティのリストを取得します
listGroups = JPT.GetAllGroups()
JPT.Debugger(listGroups) # リスト Group, サイズ = 10

# すべての「接続」エンティティのリストを取得：MPC/RBE2/..
listConnections = JPT.GetAllConnections()
JPT.Debugger(listConnections) # リスト Connect, サイズ = 2

# すべての「座標」エンティティのリストを取得します
listCoordinates = JPT.GetAllCoordinates()
JPT.Debugger(listCoordinates) # リスト Coord, サイズ = 1

# すべての「負荷と境界条件」エンティティのリストを取得する
listLBCs = JPT.GetAllLoadBoundaryConditions()
JPT.Debugger(listLBCs) # リスト LBC, サイズ = 5

# ditem形式ですべての「負荷と境界条件」エンティティのリストを取得する
listEntity = JPT.GetAllLoadsBCs()
JPT.Debugger(listEntity) # リスト DItem, サイズ = 5

############  エンティティのリストを取得してくださいテーブルタイプがあります ##############
# データベースにテーブルタイプがあるすべての「ロードケース」エンティティのリストを取得します
listEntityLoadCases = JPT.GetAllByTableTypeID(JPT.DTableType.DTABLE_LOADCASE)
JPT.Debugger(listEntityLoadCases) # リスト DItem, サイズ = 0

# データベースにテーブルタイプがあるすべての「ローカルセット」エンティティのリストを取得する
listEntityLocalSettings = JPT.GetAllByTableTypeID(JPT.DTableType.DTABLE_LOCAL_SETTING)
JPT.Debugger(listEntityLocalSettings) # リスト DItem, サイズ= 1

############ ditem/elemタイプでエンティティのリストを取得する（edge/1d要素） ##############
# すべての「エッジ」エンティティのリストをDITEMタイプで取得する
listEntityEdges = JPT.GetAllByType(JPT.DItemType.EDGE)
JPT.Debugger(listEntityEdges) # リスト DItem, サイズ = 173

# 要素の種類ごとにすべての「1D要素」のリストを取得する
list1DElems = JPT.GetElemsByKind(JPT.ElemKind.ELEMKIND_1D)
JPT.Debugger(list1DElems) # リスト Elem, サイズ = 383

# プロパティIDでマテリアルを入手してください。例：プロパティIDは1です
idProperty = 1
material = JPT.GetMaterialFromProperty(idProperty)
JPT.Debugger(material) # DItem, 名前 = Stainless_Steel

# パートのIDごとにパートに関連付けられたすべての顔のリストを取得します。例：パートIDは2です
idPart = 2
listAssociatedFaces = JPT.GetEntitiesByAssociation(JPT.DItemType.BODY, JPT.AssociateType.AS_FACE, idPart)
JPT.Debugger(listAssociatedFaces) # リスト DItemVector, サイズ = 10

############  エンティティテーブルのタイプとエンティティのプロパティによるエンティティのリストを取得する(名前、ID) ##############
# DITEMタイプとエンティティIDでエンティティオブジェクトのリストを取得します。例：要素IDは39869です
idElem = 39869
listEntityByID = JPT.GetEntitiesByID(JPT.DItemType.ELEM, idElem)
JPT.Debugger(listEntityByID) # リスト DItem, サイズ = 1

# GエンティティオブジェクトのETリスト 'Body' DTabel から始まるタイプとエンティティ名 "Block"
isMatch = JPT.BoolType.TRUE_VAL
entityName = "Block"
listEntityByName = JPT.GetEntitiesByName(JPT.DTableType.DTABLE_BODY, entityName, isMatch)
JPT.Debugger(listEntityByName) # リスト DItem, サイズ = 1

# 得る 'Node' でエンティティ x,y,z 位置
x = 0
y = 0
z = 10
listNodeByPos = JPT.GetEntitiesByPosition(JPT.AssociateType.AS_NODE, x, y, z)
JPT.Debugger(listNodeByPos) # リスト DItem, サイズ = 1
