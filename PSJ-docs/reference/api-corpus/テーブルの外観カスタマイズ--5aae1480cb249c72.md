このチュートリアルでは、テーブルの外観をカスタマイズする方法を学びます。

テキストの色: テーブルのセルのフォントの色を変更します。

from pyjdg import *
def on_menu(dlg,name,menu):
    cellvector=dlg.get_table_sel_cell(name="Table1")
    if menu=="全ての選択されたセルの塗りつぶしの色をカラーピッカーで設定":
        dlg.set_table_cell_fill_color(name="Table1")
    elif menu=="選択されたセルのみの塗りつぶしの色をカラーピッカーで設定":
         dlg.set_table_cell_fill_color(name="Table1",
            cell=cellvector[0])
    elif menu=="選択されたセルのみの塗りつぶしの色を赤に設定":
        dlg.set_table_cell_fill_color(name="Table1",
            cell=cellvector[0],color=7105764)
    elif menu=="最初のセルのみの塗りつぶしの色を赤に設定":
        dlg.set_table_cell_fill_color(name="Table1",
            cell=TableCellID(row=0,col=0),color=7105764)
    elif menu=="最初の2x2セルのみの塗りつぶしの色を赤に設定":
        [[dlg.set_table_cell_fill_color(name="Table1",
            cell=TableCellID(row=i,col=j),color=7105764)
            for i in (0,1)] for j in (0,1)]

def main():
    dlg=JDGCreator(title="ダイアログ",resizable=True,validation=True)
    dlg.add_table(name="Table1",rows=5,
        columns=["見出し1","見出し2"],
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="OK",layout="footer")
    dlg.add_button(name="ButtonCancel",text="キャンセル",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_table_right_menu(name="Table1",
        menus=["全ての選択されたセルの塗りつぶしの色をカラーピッカーで設定",
        "選択されたセルのみの塗りつぶしの色をカラーピッカーで設定",
        "選択されたセルのみの塗りつぶしの色を赤に設定",
        "最初のセルのみの塗りつぶしの色を赤に設定",
        "最初の2x2セルのみの塗りつぶしの色を赤に設定"])
    dlg.generate_window()
    dlg.on_table_right_menu(name="Table1",callfunc=on_menu)

if __name__=='__main__':
    main()
