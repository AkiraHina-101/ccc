# ファイルおよびフォルダーブラウザ

ファイルおよびフォルダーブラウザ

🎯 導入​

このチュートリアルでは、ファイルとフォルダーのブラウザを使用する方法を学習します dlg.add_browser():

ダイアログには3つのブラウザがあります。

デフォルトのパス「C：\プログラムファイル（x86）」を備えたファイルブラウザ。

フォルダーブラウザ。

拡張機能フィルターを使用してブラウザをファイルします。

📖 チュートリアル​

# 日本語のエンコーディング
# コーディング: cp932

from pyjdg import *

def main():
    dlg=JDGCreator(title="ブラウザサンプル",resizable=True,validation=True)
    dlg.add_groupbox(name="GroupBox1",text="ファイルブラウザ",layout="Window")
    dlg.add_label(name="Label2",text="デフォルトパスを備えたファイルブラウザ",layout="GroupBox1")
    dlg.add_browser(name="Open File/Folder3",mode="file",file_filter="All Files(*.*)", \
        default="C:\Program Files (x86)",layout="GroupBox1")
    dlg.add_groupbox(name="GroupBox4",text="フォルダーブラウザ",layout="Window")
    dlg.add_browser(name="Open File/Folder5",mode="folder",file_filter="All Files(*.*)",layout="GroupBox4")
    dlg.add_groupbox(name="GroupBox6",text="フィルター付きのファイルブラウザ",layout="Window")
    dlg.add_label(name="Label7",text="フィルター形式: description(extension;extension). " \
        "For example: InpFiles(*.inp), txt files(*.txt;*.text)",layout="GroupBox6")
    dlg.add_browser(name="Open File/Folder8",mode="file",
        file_filter="Inp Files(*.inp),Txt Files(*.txt;*.text), All Files(*.*)",layout="GroupBox6")
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="わかりました",layout="footer")
    dlg.add_button(name="ButtonCancel",text="キャンセル",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()

if __name__=='__main__':
    main()
