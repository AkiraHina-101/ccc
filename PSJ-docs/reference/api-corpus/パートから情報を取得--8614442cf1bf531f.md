# パートから情報を取得

パートから情報を取得

🎯 はじめに​

本チュートリアルでは、パートからデータベース情報を取得する方法を紹介します:

インストールされているJupiterのパスを取得し、サンプルモデルをJupiterに読み込む。

すべてのパートリストの最初のパート情報を取得し、変数'part'に代入する。その後、Python APIウィンドウにpartの名前とIDを表示する。

partの名前/サーフェス色/メッシュライン色/透明度/を変更する。

partのエンティティ(フェース/エッジ/頂点/要素/節点)の数を取得する。

partの材料やプロパティを取得する。

partに関連する荷重および境界条件/接続/ローカルメッシュ設定/接触を取得する。

partの面積/体積/質量/重心/慣性モーメントを取得する。

📖 チュートリアル​

# Python APIウィンドウ内のログメッセージを消去します。
JPT.ClearLog()

# インストールされているJupiterとサンプルデータのパスを取得します。
#　その後サンプルモデルをJupiterに読み込みます。
jupiterPath = JPT.GetAppPathInfo(JPT.PathType.PROGRAM_PATH)
samplePath = jupiterPath + '/SampleData/PSJ/PSJ-Utility/JtdbSample/Get_Infor_Part_Sample.jtdb'
FileMenu.LoadJTDB(strFileName=samplePath)

# モデルをメインウィンドウのサイズに合わせて表示します。
JPT.ViewFitToModel()

# 現在のモデルのすべてのパートのリストを取得します。
listpart = JPT.GetAllParts()

# すべてのパートリストの中から最初のパートの情報を取得し、変数'part'に代入します。
# その後、Python APIウィンドウにパートのIDと名前が表示されます。
part = listpart[0]
JPT.Debugger(part)
print (part.id)
print (part.name)

# 'part'に代入している名前を変更します。
part.name = "change_name_of_first_part"

# 'part'のパート色をランダムな色に変更します。
part.color = JPT.GetRandomJPTColor()

# 'part'のパート色をRGBカラーコードで変更します。
part.color = JPT.ConvertRGBToJPTColor(255,0,0)

# 'part'のメッシュラインの色をRGBカラーコードで変更します。
part.colorMesh = JPT.ConvertRGBToJPTColor(255,255,255)

# 'part'の透明度を50%に設定します。
part.transparency = 50

# 'part'からすべてのフェースを取得します。
listFaces = part.faces
JPT.Debugger(listFaces)

# 'part'からすべての要素を取得します。
listElems = part.elems
JPT.Debugger(listElems)

# 'part'からすべての節点を取得します。
listNodes = part.nodes
JPT.Debugger(listNodes)

# 'part'からすべてのエッジを取得します。
listEdges = part.edges
JPT.Debugger(listEdges)

# 'part'からすべての頂点を取得します。
listVertex = part.vertex
JPT.Debugger(listVertex)

# 'part'から材料を取得します。
# その後、Python APIウィンドウに材料のIDと名前を表示します。
listMaterials = part.materials
if len(listMaterials) == 0:
    print ("このパートには材料が割り当てられていません。")
else:
    for material in listMaterials:
        print ("'part'に割り当てられている材料のID: " + str(material.id))
        print ("'part'に割り当てられている材料の名前: " + str(material.name))

# 'part'からプロパティを取得します。
# その後、Python APIウィンドウにプロパティのIDと名前を表示します。
listProperties = part.properties
if len(listProperties) == 0:
    print ("このパートにはプロパティが割り当てられていません。")
else:
    for property in listProperties:
        print ("'part'に割り当てられているプロパティのID: " + str(property.id))
        print ("'part'に割り当てられているプロパティの名前: " + str(property.name))

# 'part'に関連するすべての荷重と境界条件を取得します。
# その後、Python APIウィンドウに荷重と境界条件のIDと名前を表示します。 
listLoadAndBCs = part.loadAndBCs
if len(listLoadAndBCs) == 0:
    print ("このパートには関連する荷重と境界条件がありません。")
else:
    for loadAndBC in listLoadAndBCs:
        print ("'part'に関連する荷重および境界条件のID: " + str(loadAndBC.id))
        print ("'part'に関連する荷重および境界条件の名前: " + str(loadAndBC.name))

# 'part'に関連するすべての接続を取得します。
# その後、Python APIウィンドウに接続のIDと名前を表示します。
listConnections = part.connections
if len(listConnections) == 0:
    print ("このパートには関連する接続要素(MPC/RBE2)がありません。")
else:
    for connection in listConnections:
        print ("'part'に関連する接続のID:  " + str(connection.id))
        print ("'part'に関連する接続の名前: " + str(connection.name))

# 'part'からすべてのローカルメッシュ設定を取得します。
# その後、Python APIウィンドウにローカルメッシュ設定のIDと名前を表示します。
listLocalMeshSettings = part.localMeshSettings
if len(listLocalMeshSettings) == 0:
    print ("このパートにはローカルメッシュ設定がありません。")
else:
    for localMeshSetting in listLocalMeshSettings:
        print ("'part'のローカルメッシュ設定のID: " + str(localMeshSetting.id))
        print ("'part'のローカルメッシュ設定の名前: " + str(localMeshSetting.name))

# 'part'に関連するすべての接触を取得します。
# その後、Python APIウィンドウに接触のIDと名前を表示します。 
listContacts = part.contacts
if len(listContacts) == 0:
    print ("このパートには関連する接触がありません。")
else:
    for contact in listContacts:
        print ("'part'に関連する接触のID: " + str(contact.id))
        print ("'part'に関連する接触の名前: " + str(contact.name))

# 'part'の面積を取得し、Python APIウィンドウに表示します。
print("'part'の面積: " + str(part.area))

# 'part'の体積を取得し、Python APIウィンドウに表示します。
print("'part'の体積: " + str(part.volume))

# 'part'の質量を取得し、Python APIウィンドウに表示します。
print("'part'の質量: " + str(part.mass))

#  'part'の重心の値を取得し、Python APIウィンドウにに表示します。
print("'part'の重心: [%s %s %s] " % \
    (str(part.gravityCenter.x) , str(part.gravityCenter.y),  str(part.gravityCenter.z)))

# 'part'の慣性モーメントを取得し、Python APIウィンドウに表示します。
print("'part'の慣性モーメント: " )
JPT.Debugger(part.momentInertia)
