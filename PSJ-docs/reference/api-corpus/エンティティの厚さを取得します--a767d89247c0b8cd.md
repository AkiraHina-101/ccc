# エンティティの厚さを取得します

エンティティの厚さを取得します

🎯 導入​

このチュートリアルでは、エンティティ（ パート/フェイス/要素）から厚さを取得する方法を学びます。

Face ID 52からすべての2D要素のリストを取得します。

財産の資料を作成します。

異なる厚さ値で3つのシェルプロパティを作成します。

エンティティのIDから厚さを取得します。

📖 チュートリアル​

# Python APIウィンドウでログメッセージをクリアします
JPT.ClearLog()

# 2つのキューブパートを作成します
cube1 = Geometry.Part.Cube(iPartColor=12999622)
cube2 = Geometry.Part.Cube(dlOrigin=[0.015, 0.0, 0.0], strName="Cube_2", iPartColor=5093709)

# ディスプレイビューのモデルのビューをフィットします。
JPT.ViewFitToModel()

# FaceID52からすべての2D要素のリストを取得する
elem_from_FaceID52 = MainWindow.RightClick.AssociatedPick(crlInput=[Face(52)], strTarget="Elem")

# 素材を作成します
Properties.Material.Add("Copper_Alloy", [Density([(DENSITY, 8.3e-09)]), Elastic([(YOUNGS_MODULUS, 110000.0),
    (POISSONS_RATIO, 0.34)])])

############ 異なる厚さ値で3つのシェルプロパティを作成します############
# 最初のシェルプロパティはCube1用です
Properties.Shell(strName="Cube1のシェルプロパティ1", iPropertyId=1, dThickness=0.005,
    crMatMembrane=Material(1), crMatBend=Material(1), crMatShear=Material(1), crlTargets=[cube1])

# セカンドシェルプロパティは、フェイスID 47,48,49,50,51用です
Properties.Shell(strName="フェイスID 47-51のシェルプロパティ2", iPropertyId=2, dThickness=0.008,
    crMatMembrane=Material(1), crMatBend=Material(1), crMatShear=Material(1), crlTargets=[Face(47, 48, 49, 50, 51)])

# 3番目のシェルプロパティは、顔ID52のすべての2D要素向けです
Properties.Shell(strName="フェイスID52のすべての2D要素のシェルプロパティ3", iPropertyId=3, dThickness=0.012,
    crMatMembrane=Material(1), crMatBend=Material(1), crMatShear=Material(1), crlTargets=elem_from_FaceID52)

######################## エンティティから厚さを取得します########################
# IDから厚さを取得します 'Body' 実在物
dThicknessBody_ID1 = JPT.GetThicknessOfEntity(JPT.DItemType.BODY, 1)
print("の厚さ 'Body' エンティティID 1: {}".format(dThicknessBody_ID1))

# IDから厚さを取得します 'Face' 実在物
dThicknessFace_ID49 = JPT.GetThicknessOfEntity(JPT.DItemType.FACE, 49)
print("の厚さ'Face' エンティティID 49: {}".format(dThicknessFace_ID49))

# IDから厚さを取得します'Element' 実在物
dThicknessElem_ID1477 = JPT.GetThicknessOfEntity(JPT.DItemType.ELEM, 1477)
print("2Dの厚さ 'Element' エンティティID 1477: {}".format(dThicknessElem_ID1477))

dThicknessElem_ID2114 = JPT.GetThicknessOfEntity(JPT.DItemType.ELEM, 2114)
print("2Dの厚さ 'Element' エンティティID 2114: {}".format(dThicknessElem_ID2114))
