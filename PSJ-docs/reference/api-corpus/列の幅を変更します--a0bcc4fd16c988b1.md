列の幅を変更します

🎯 導入​

このチュートリアルでは、テーブル内のセルの幅を変更/サイズ変更する方法を学びます。

新しいテーブルを作成します。

右クリックしてコンテキストメニューを使用して、セルの幅を制御します。

📖 チュートリアル​

# 日本語のエンコーディング
# コーディング: cp932

from pyjdg import *
def on_menu(dlg,name,menu):
    cellvector=dlg.get_table_sel_cell(name)
    if menu=="ダイアログを使用して列幅を設定します":
        dlg.set_table_column_width(name="Table1",
            col=cellvector[0].col_number)
    if menu=="列の幅を400等に設定します":
        dlg.set_table_column_width(name="Table1",
            col=cellvector[0].col_number,width=400)
    elif menu=="列幅を取得します":
        col_width = dlg.get_table_column_width(name="Table1",
            col=cellvector[0].col_number)
        print("カラム "+str(cellvector[0].col_number)+
            " 幅はです: "+str(col_width))

def main():
    dlg=JDGCreator(title="ダイアログ",resizable=True,validation=True)
    dlg.add_table(name="Table1",rows=5,
        columns=["Heading1","Heading2"],
        layout="Window",width=260,height=260)
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="わかりました",layout="footer")
    dlg.add_button(name="ButtonCancel",text="キャンセル",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_table_right_menu(name="Table1",
        menus=["ダイアログを使用して列幅を設定します",
        "列の幅を400等に設定します","列幅を取得します"])
    dlg.generate_window()
    dlg.on_table_right_menu(name="Table1",callfunc=on_menu)

if __name__=='__main__':
    main()
