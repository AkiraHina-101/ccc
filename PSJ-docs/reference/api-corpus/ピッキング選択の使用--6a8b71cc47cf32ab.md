# coding: cp932

from pyjdg import *

def display_selected_entities(dlg, sel_list):
    selected = JPT.GetSelectedNodes()
    if selected.sizeVec()>0:
        str_node=''
        for n in sel_list:
            str_node += '節点:' + str(n.id) + ', '
        dlg.set_item_text(name="node",text=str_node)
    else:
        dlg.set_item_text(name="node",text="")

def main():
    dlg=JDGCreator(title="ピッキングサンプル",resizable=True,validation=True)
    dlg.add_groupbox(name="GroupBox1",text="グループボックス",layout="Window")
    dlg.add_label(name="infor1",text="このサンプルはアイテムをピックしたときに、",layout="GroupBox1")
    dlg.add_label(name="infor2",text="独自の関数でそれをすぐに受け取る方法を示します。",layout="GroupBox1")
    dlg.add_label(name="infor3",text="その関数は2つのパラメータを持つ必要があります。1つはdlg、もう1つは選択によって返されるリストです。",layout="GroupBox1")
    dlg.add_label(name="infor4",text="例：def displaySelectedEntity(dlg,selList):",layout="GroupBox1")
    dlg.add_groupbox(name="GroupBox2",text="ここに選択された節点が表示されます",layout="Window")
    dlg.add_richeditbox(name="node",width=100,height=200,layout="GroupBox2")
    dlg.add_node_selector()
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonOk",text="OK",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()
    dlg.on_dlg_selector_changed(callfunc=display_selected_entities)

if __name__=='__main__':
    main()

# coding: cp932

from pyjdg import *

def displace_selected_entities(dlg):
    str_nodes = ""
    for n in dlg.get_selected_entities(0):
        str_nodes += '選択された節点ID: ' + (str(n.id)) + ',\n'
    str_nodes = str_nodes[:-2]
    dlg.set_item_text(name="selected_nodes",text=str_nodes)

def main():
    dlg=JDGCreator(title="ピッキングサンプル",resizable=True,validation=True)
    dlg.add_groupbox(name="GroupBox1",text="選択されたすべての節点を表示",layout="Window")
    dlg.add_label(name="Label2",text="節点を選択して「節点を表示」ボタンをクリックしてください",layout="GroupBox1")
    dlg.add_button(name="display_nodes",text="節点を表示",layout="GroupBox1")
    dlg.add_richeditbox(name="selected_nodes",text="",width=200,height=200,layout="GroupBox1")
    dlg.add_node_selector()
    dlg.add_hlayout(name="footer",layout="Window")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.add_button(name="ButtonCancel",text="Cancel",layout="footer")
    dlg.add_space(orientation="horizontal",layout="footer")
    dlg.generate_window()
    dlg.on_button_clicked(name="display_nodes",callfunc=displace_selected_entities)

if __name__=='__main__':
    main()
